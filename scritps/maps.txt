        let map, markers = [], ghostMarkers = [];
        const CUBA_CENTER = [21.5218, -77.7812];
        const CUBA_ZOOM = 6;

        function normalizar(texto) {
            if (!texto) return "";
            return texto.toString().trim().normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase().replace(/\s+/g, ' ');
        }

        function getCoordenadas(prov, mun) {
            if (!dbGeografia[prov]) return null;
            const munNorm = normalizar(mun);
            const found = dbGeografia[prov].find(m => normalizar(m.nombre) === munNorm);
            return found ? { lat: found.lat, lng: found.lng } : null;
        }

        function initMap() {
            map = L.map('map', { scrollWheelZoom: false }).setView(CUBA_CENTER, CUBA_ZOOM);
            L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
                attribution: '© OpenStreetMap'
            }).addTo(map);

            setupFilters();
            applyFilters();
        }

        function setupFilters() {
            const provSelect = document.getElementById('filter-provincia');
            Object.keys(dbGeografia).sort().forEach(p => {
                const opt = document.createElement('option');
                opt.value = p;
                opt.textContent = p;
                provSelect.appendChild(opt);
            });
        }

        function updateMunicipios() {
            const prov = document.getElementById('filter-provincia').value;
            const munSelect = document.getElementById('filter-municipio');
            munSelect.innerHTML = '<option value="">Todos los municipios</option>';
            if (prov && dbGeografia[prov]) {
                dbGeografia[prov].map(m => m.nombre).sort().forEach(m => {
                    const opt = document.createElement('option');
                    opt.value = m; opt.textContent = m;
                    munSelect.appendChild(opt);
                });
            }
        }

        function applyFilters() {
            const prov = document.getElementById('filter-provincia').value;
            const mun = document.getElementById('filter-municipio').value;
            const inst = document.getElementById('filter-inst').value;

                return (prov === "" || r.provincia === prov) &&
                       (mun === "" || normalizar(r.nombre) === normalizar(mun)) &&
                       (inst === "" || r.inst === inst);
            });

            renderMarkers(filtered, prov);
            renderList(filtered);
            updateStats(filtered.length);

            // Ajustar cámara
            if (mun !== "") {
                const c = getCoordenadas(prov, mun);
                if (c) map.flyTo([c.lat, c.lng], 13);
            } else if (prov !== "") {
                const group = new L.featureGroup([...markers, ...ghostMarkers]);
                if (group.getLayers().length > 0) map.fitBounds(group.getBounds(), { padding: [30, 30] });
            } else {
                map.flyTo(CUBA_CENTER, CUBA_ZOOM);
            }
        }

        function renderMarkers(data, provSel) {
            markers.forEach(m => map.removeLayer(m));
            ghostMarkers.forEach(m => map.removeLayer(m));
            markers = [];
            ghostMarkers = [];

            // 1. Marcadores ROJOS (Con datos)
            data.forEach(r => {
                const c = getCoordenadas(r.provincia, r.nombre);
                if (c) {
                    const redIcon = L.divIcon({
                        className: 'custom-div-icon',
                        html: `<div class="marker-dot bg-filled"></div>`,
                        iconSize: [14, 14],
                        iconAnchor: [7, 7]
                    });

                    const m = L.marker([c.lat, c.lng], { icon: redIcon })
                        .addTo(map)
                        .bindPopup(`
                            <div class="p-1">
                                <h4 class="font-bold text-primary">${r.inst}</h4>
                                <p class="text-xs text-gray-600">${r.nombre}, ${r.provincia}</p>
                                <span class="text-[10px] bg-red-50 text-red-700 px-2 py-1 rounded mt-2 block border border-red-100">Datos registrados</span>
                            </div>
                        `)
                        .bindTooltip(r.nombre, { direction: 'top', className: 'custom-tooltip' });
                    markers.push(m);
                }
            });

            // 2. Marcadores GRISES (Municipios de la provincia sin datos)
            if (provSel !== "") {
                const munConDatos = data.map(d => normalizar(d.nombre));
                dbGeografia[provSel].forEach(geo => {
                    if (!munConDatos.includes(normalizar(geo.nombre))) {
                        const grayIcon = L.divIcon({
                            className: 'custom-div-icon',
                            html: `<div class="marker-dot bg-empty"></div>`,
                            iconSize: [12, 12],
                            iconAnchor: [6, 6]
                        });
                        const gm = L.marker([geo.lat, geo.lng], { icon: grayIcon })
                            .addTo(map)
                            .bindTooltip(`${geo.nombre} (Pendiente)`, { direction: 'top', className: 'custom-tooltip tooltip-empty' });
                        ghostMarkers.push(gm);
                    }
                });
            }
        }

        function renderList(data) {
            const list = document.getElementById('municipios-list');
            list.innerHTML = '';
            if (data.length === 0) {
                list.innerHTML = '<div class="col-span-full py-10 text-center text-gray-400">Sin resultados.</div>';
                return;
            }
            data.forEach(r => {
                const c = getCoordenadas(r.provincia, r.nombre);
                const item = document.createElement('div');
                item.className = 'bg-white p-4 border rounded-xl card-hover cursor-pointer border-l-4 border-l-primary transition-all';
                item.onclick = () => { if (c) map.flyTo([c.lat, c.lng], 14); };
                item.innerHTML = `
                    <div class="text-[10px] text-gray-400 font-bold uppercase mb-1">${r.provincia}</div>
                    <h4 class="font-bold text-gray-800 mb-1">${r.inst}</h4>
                    <p class="text-xs text-gray-500 mb-2">${r.nombre}</p>
                    <div class="flex flex-wrap border-t pt-2 mt-2">
                        ${r.necesidades.map(n => `<span class="badge-need">${n}</span>`).join('')}
                    </div>
                `;
                list.appendChild(item);
            });
        }

        function updateStats(n) {
            document.getElementById('stats').innerHTML = `Se visualizan <b class="text-primary">${n}</b> instituciones con diagnósticos completados.`;
        }

        function resetFilters() {
            document.getElementById('filter-provincia').value = '';
            updateMunicipios();
            document.getElementById('filter-inst').value = '';
            applyFilters();
        }

        window.onload = initMap;
